particle.js JavaScript 
// 粒子会被光标推开 —— 就这么几行
function repel(p, mouse, R = 120) {
  const dx = p.x - mouse.x;
  const dy = p.y - mouse.y;
  const d  = Math.hypot(dx, dy) || 1;

  if (d < R) {
    const f = (1 - d / R) * 2.4;
    p.vx += (dx / d) * f;
    p.vy += (dy / d) * f;
  }
}

// 每帧调用一次，剩下的交给物理
for (const p of particles) {
  repel(p, pointer);
  p.x += p.vx;
  p.y += p.vy;
}
